  CREATE TABLE IF NOT EXISTS acl (
   "id" bigserial PRIMARY KEY,
   "class" text NOT NULL REFERENCES acl_class("name"),
   "user" text REFERENCES app_user(username),
   "group" text REFERENCES acl_group("name"),
   "create" BOOLEAN NOT NULL DEFAULT FALSE,
   "read" BOOLEAN NOT NULL DEFAULT FALSE,
   "update" BOOLEAN NOT NULL DEFAULT FALSE,
   "delete" BOOLEAN NOT NULL DEFAULT FALSE,
   "statement" JSONB, 
   "remark" text,
   UNIQUE ("class", "user")
   -- PRIMARY KEY ("class", "user")
  );

#+begin_src sql :tangle acl.sql
  CREATE OR REPLACE FUNCTION get_acl("class" text, "user" text, args JSON DEFAULT '{}')
   RETURNS acl LANGUAGE SQL AS $$
   SELECT * FROM acl WHERE "class" = $1 AND "user" = $2 OR "user" IS NULL ORDER BY "user" NULLS LAST LIMIT 1;
   $$;
#+end_src
